# Finds the player's car speed in memory: samples every float in a plausible range at several
# speeds, with a screenshot of the speedometer at each moment. Then:
#   python tools/memscan_match.py <build_logs\autoplay-run folder> s1=MPH s2=MPH ...
# with the speeds read from the screenshots (test-<run>-s1.png ...).
include common.txt

do boot_to_festival
wait 5
set rt
wait 3
memscan_start 0.5 400
memscan_sample s1
shot s1
wait 1.5
memscan_sample s2
shot s2
wait 1.5
memscan_sample s3
shot s3
clear
hold lt 1.2
memscan_sample s4
shot s4
wait 4
memscan_sample s5
shot s5
quit
