# Finds the player car's world position: floats that stay exactly the same while the car is
# stopped (traffic and other moving things fail) and change, but by less than the distance the car
# could travel, while it drives (the static world fails). Forward and backward runs.
include common.txt

def stopped_check
  wait 1.5
  memscan_filter same
  wait 1
  memscan_filter same
end

def drive_check
  wait 2
  memscan_filter delta_gt 0.05
  memscan_filter delta_lt 80
end

do boot_to_festival
wait 3
hold a 2
wait 1
memscan_start 0.5 20000
do stopped_check
set rt
do drive_check
clear
hold a 4
do stopped_check
set lt
do drive_check
clear
hold a 4
do stopped_check
set rt
do drive_check
clear
hold a 4
do stopped_check
memscan_filter abs_gt 1
memscan_list 20
memscan_ptrs 0x1000 3
set rt
wait 1
clear
hold a 4
quit
