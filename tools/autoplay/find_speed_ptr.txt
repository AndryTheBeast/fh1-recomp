# Car-speed calibration: repeated cycles of accelerate (speed rises), a short handbrake pull while
# still sliding (car speed stays high, locked rear wheels drop to ~0) and handbrake to a full stop
# (speed ~0). A = handbrake. No brake trigger: holding LT after stopping reverses the car.
include common.txt

def cycle
  set rt
  wait 1.5
  memscan_filter inc
  wait 1
  memscan_filter inc
  memscan_filter abs_gt 4
  clear
  hold a 0.5
  memscan_filter abs_gt 3
  hold a 5
  wait 1
  memscan_filter abs_lt 0.3
end

do boot_to_festival
wait 5
set rt
wait 2.5
memscan_start 3 300
wait 1
memscan_filter inc
wait 1
memscan_filter inc
clear
hold a 0.5
memscan_filter abs_gt 4
hold a 5
wait 1
memscan_filter abs_lt 0.3
do cycle x2
memscan_list 80
set rt
wait 2
memscan_list 80
shot moving
clear
hold a 5
quit
